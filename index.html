-- never trust the client
Remote.OnServerEvent:Connect(function(plr, id)
  if typeof(id) ~= "string" then return end
  local a = Abilities[id]
  if not a or not Owns(plr, id) then return end
  if Cooldown:Active(plr, id) then return end
  a:Execute(plr)
end)
